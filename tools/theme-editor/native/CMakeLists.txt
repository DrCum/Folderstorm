  set(THEME_HOST_GENERATED_INCLUDE_DIR "" CACHE PATH "Generated native headers for an optional compile-only host check")

    target_include_directories(host_selection_compile SYSTEM PRIVATE ../../session-host ../../../indra/newview "${THEME_BOOST_INCLUDE_DIR}" "${THEME_HOST_GENERATED_INCLUDE_DIR}")
    target_compile_definitions(host_selection_compile PRIVATE UNICODE _UNICODE NOMINMAX)
    if(MSVC)
      target_compile_options(host_selection_compile PRIVATE /utf-8 /permissive-)
    endif()
  endif()
  add_executable(host_preview_checks host_preview_checks.cpp ../../session-host/host_preview.cpp ../../session-host/host_theme.cpp ../../session-host/accessibility.cpp)
  target_include_directories(host_preview_checks SYSTEM PRIVATE ../../session-host)
  target_link_libraries(host_preview_checks PRIVATE theme_contract user32 gdi32 msimg32 advapi32 ole32 uuid oleacc comctl32 uxtheme shell32)
  target_compile_definitions(host_preview_checks PRIVATE UNICODE _UNICODE NOMINMAX)
  if(MSVC)
    target_compile_options(host_preview_checks PRIVATE /W4 /WX /utf-8 /permissive-)
  endif()
  add_executable(host_preview_probe WIN32 host_preview_probe.cpp ../../session-host/host_preview.cpp ../../session-host/host_theme.cpp ../../session-host/accessibility.cpp)
  target_include_directories(host_preview_probe SYSTEM PRIVATE ../../session-host)
  target_link_libraries(host_preview_probe PRIVATE theme_contract user32 gdi32 msimg32 advapi32 ole32 uuid oleacc comctl32 uxtheme shell32)
  target_compile_definitions(host_preview_probe PRIVATE UNICODE _UNICODE NOMINMAX)
  if(MSVC)
    target_compile_options(host_preview_probe PRIVATE /W4 /WX /utf-8 /permissive-)
  endif()
  target_link_libraries(theme_contract PUBLIC bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(theme_contract PUBLIC OpenSSL::Crypto)
endif()

  target_compile_options(theme_contract PRIVATE /W4 /WX /utf-8 /permissive-)

  target_compile_options(theme_contract PRIVATE -Wall -Wextra)
