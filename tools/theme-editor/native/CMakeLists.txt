cmake_minimum_required(VERSION 3.20)
project(FolderstormThemeContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_path(THEME_BOOST_INCLUDE_DIR boost/json.hpp REQUIRED)
include("${CMAKE_CURRENT_SOURCE_DIR}/ThemeRules.cmake")
folderstorm_theme_rules("${CMAKE_CURRENT_SOURCE_DIR}/../contracts/xui-edit-rules.json")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/xui-rules-sha256.txt" "${THEME_RULES_SHA256}")
configure_file(fsthemerules.h.in fsthemerules.h @ONLY)
add_library(theme_contract STATIC ../../../indra/newview/fsthemepackage.cpp boost_json.cpp)
target_include_directories(theme_contract PUBLIC ../../../indra/newview "${THEME_BOOST_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
if(WIN32)
  add_executable(host_preview_checks host_preview_checks.cpp ../../session-host/host_preview.cpp ../../session-host/host_theme.cpp ../../session-host/accessibility.cpp)
  target_include_directories(host_preview_checks SYSTEM PRIVATE ../../session-host)
  target_link_libraries(host_preview_checks PRIVATE theme_contract user32 gdi32 msimg32 advapi32 ole32 uuid oleacc comctl32 uxtheme shell32)
  target_compile_definitions(host_preview_checks PRIVATE UNICODE _UNICODE NOMINMAX)
  if(MSVC)
    target_compile_options(host_preview_checks PRIVATE /W4 /WX /utf-8 /permissive-)
  endif()
  add_executable(host_preview_probe WIN32 host_preview_probe.cpp ../../session-host/host_preview.cpp ../../session-host/host_theme.cpp ../../session-host/accessibility.cpp)
  target_include_directories(host_preview_probe SYSTEM PRIVATE ../../session-host)
  target_link_libraries(host_preview_probe PRIVATE theme_contract user32 gdi32 msimg32 advapi32 ole32 uuid oleacc comctl32 uxtheme shell32)
  target_compile_definitions(host_preview_probe PRIVATE UNICODE _UNICODE NOMINMAX)
  if(MSVC)
    target_compile_options(host_preview_probe PRIVATE /W4 /WX /utf-8 /permissive-)
  endif()
  target_link_libraries(theme_contract PUBLIC bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(theme_contract PUBLIC OpenSSL::Crypto)
endif()
if(MSVC)
  target_compile_options(theme_contract PRIVATE /W4 /WX /utf-8 /permissive-)
else()
  target_compile_options(theme_contract PRIVATE -Wall -Wextra)
endif()
add_executable(theme_validate main.cpp)
target_link_libraries(theme_validate PRIVATE theme_contract)
add_executable(host_selection_checks host_selection_checks.cpp ../../session-host/host_theme.cpp)
target_include_directories(host_selection_checks PRIVATE ../../session-host)
target_link_libraries(host_selection_checks PRIVATE theme_contract)
if(MSVC)
  target_compile_options(host_selection_checks PRIVATE /W4 /WX /utf-8 /permissive-)
endif()
if(WIN32)
  add_executable(host_gdi_checks host_gdi_checks.cpp ../../session-host/host_theme.cpp ../../session-host/accessibility.cpp)
  target_include_directories(host_gdi_checks SYSTEM PRIVATE ../../session-host)
  target_link_libraries(host_gdi_checks PRIVATE theme_contract user32 gdi32 msimg32 advapi32 ole32 uuid oleacc comctl32 uxtheme)
  target_compile_definitions(host_gdi_checks PRIVATE UNICODE _UNICODE NOMINMAX)
  if(MSVC)
    target_compile_options(host_gdi_checks PRIVATE /W4 /WX /utf-8 /permissive-)
  endif()
endif()
