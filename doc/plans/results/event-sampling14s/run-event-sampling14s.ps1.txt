param(
    [Parameter(Mandatory=$true)][ValidateSet('direct','owned','default','event')][string]$Mode,
    [Parameter(Mandatory=$true)][ValidateRange(1,32)][int]$Avatars,
    [Parameter(Mandatory=$true)][ValidateRange(0,2000)][int]$Gap,
    [Parameter(Mandatory=$true)][ValidateRange(1,2)][int]$Round,
    [ValidateRange(1,3)][int]$Attempt=1,
    [ValidateSet('normal','busy','closed')][string]$Load='normal',
    [switch]$Allocations
)
$ErrorActionPreference='Stop'
$workspaceRoot=Split-Path -Parent $PSScriptRoot
$kind=if($Allocations){'alloc'}else{'timing'}
$label="event14s-$kind-r$Round-$Mode-a$Avatars-g$Gap-$Load"
if($Attempt -gt 1){$label+="-attempt$Attempt"}
$recordPath=Join-Path $PSScriptRoot "$label.json"
$outputPath=Join-Path $PSScriptRoot "$label.csv"
if((Test-Path -LiteralPath $recordPath) -or (Test-Path -LiteralPath $outputPath)){throw "Retain existing observation: $label"}
function Get-IdleObservation {
    $state=Get-Content -LiteralPath (Join-Path $workspaceRoot 'performance-state.json') -Raw | ConvertFrom-Json
    $monitor=Get-Content -LiteralPath (Join-Path $workspaceRoot 'performance-build-monitor.json') -Raw | ConvertFrom-Json
    $worker=Get-Content -LiteralPath (Join-Path $workspaceRoot 'performance-worker-checkpoint.json') -Raw | ConvertFrom-Json
    $processes=@(Get-Process -ErrorAction SilentlyContinue | Where-Object {
        $_.ProcessName -match '^(cl|link|MSBuild|cmake|ninja|vpk)$|^(event-sampling14s|worker-policy14q|threadpool-scheduler|worker-wait)' } |
        Select-Object Id,ProcessName,StartTime,CPU)
    [ordered]@{utc=[DateTime]::UtcNow.ToString('o');activeBuild=$state.activeBuild;phase=$monitor.phase;
        workerStatus=$worker.status;workerCurrent=$worker.current;buildOrProbeProcesses=$processes;
        idle=(!$state.activeBuild -and !$worker.current -and $worker.status -eq 'idle' -and
            $monitor.phase -eq 'IDLE_MONITORING' -and $processes.Count -eq 0)}
}
$exe=Join-Path $PSScriptRoot $(if($Allocations){'event-sampling14s-alloc-final.exe'}else{'event-sampling14s-final.exe'})
$record=[ordered]@{case=$label;kind=$kind;round=$Round;attempt=$Attempt;mode=$Mode;avatars=$Avatars;gapUs=$Gap;load=$Load;frames=64;
    before=(Get-IdleObservation);after=$null;status='before_check';exit=$null;
    exeSha256=(Get-FileHash -LiteralPath $exe -Algorithm SHA256).Hash;
    scope='Private generated cohort and production wait selection; explicit synthetic gap; no live viewer/deadline/FPS claim';
    limits='Only before/after full-host build/probe idle; desktop load, power, timer policy uncontrolled'}
$record | ConvertTo-Json -Depth 10 | Set-Content -LiteralPath $recordPath
if(!$record.before.idle){throw "Build/probe ownership prevents measurement: $label"}
$record.status='running';$record | ConvertTo-Json -Depth 10 | Set-Content -LiteralPath $recordPath
$argument=if($Allocations){'--allocations'}else{'--benchmark'}
& $exe $argument $Mode $Avatars $Gap 64 $Load > $outputPath 2>&1
$record.exit=$LASTEXITCODE;$record.after=Get-IdleObservation
$record.status=if($record.exit -ne 0){'probe_failed'}elseif(!$record.after.idle){'overlap_exclude'}else{'completed'}
$record | ConvertTo-Json -Depth 10 | Set-Content -LiteralPath $recordPath
if($record.exit -ne 0){throw "Probe failed: $label"}
if(!$record.after.idle){throw "Build ownership changed; retain/exclude case: $label"}
[pscustomobject]@{case=$label;status=$record.status}
