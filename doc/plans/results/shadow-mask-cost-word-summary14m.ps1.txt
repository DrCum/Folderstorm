$ErrorActionPreference='Stop'
$results=Join-Path (Split-Path -Parent $PSScriptRoot) 'viewer-performance/doc/plans/results'
function Median($values){$s=@($values|Sort-Object);$n=$s.Count;if($n%2){return [double]$s[[int]($n/2)]};return ([double]$s[$n/2-1]+[double]$s[$n/2])/2}
$data=@{}
foreach($v in 'parent','current'){foreach($r in 'r1','r2'){$data["$v-$r"]=@(Import-Csv -LiteralPath (Join-Path $results "shadow-mask-cost-word-$v-$r-14m.csv"))}}
$summary=foreach($group in ($data['parent-r1']|Group-Object joints,motions,workload,enabled)){
 $c=$group.Group[0];$selected=@{}
 foreach($v in 'parent','current'){foreach($r in 'r1','r2'){$selected["$v-$r"]=@($data["$v-$r"]|Where-Object {$_.joints -eq $c.joints -and $_.motions -eq $c.motions -and $_.workload -eq $c.workload -and $_.enabled -eq $c.enabled})}}
 $p=@($selected['parent-r1'])+@($selected['parent-r2']);$n=@($selected['current-r1'])+@($selected['current-r2'])
 $pns=Median ($p|ForEach-Object {[double]$_.elapsed_ns/[double]$_.updates});$nns=Median ($n|ForEach-Object {[double]$_.elapsed_ns/[double]$_.updates})
 $pc=Median ($p|ForEach-Object {[double]$_.thread_cycles/[double]$_.updates});$nc=Median ($n|ForEach-Object {[double]$_.thread_cycles/[double]$_.updates})
 $pn1=Median ($selected['parent-r1']|ForEach-Object {[double]$_.elapsed_ns/[double]$_.updates});$pn2=Median ($selected['parent-r2']|ForEach-Object {[double]$_.elapsed_ns/[double]$_.updates})
 $pc1=Median ($selected['parent-r1']|ForEach-Object {[double]$_.thread_cycles/[double]$_.updates});$pc2=Median ($selected['parent-r2']|ForEach-Object {[double]$_.thread_cycles/[double]$_.updates})
 [pscustomobject]@{joints=$c.joints;motions=$c.motions;workload=$c.workload;enabled=$c.enabled;parent_ns_per_update=[math]::Round($pns,3);current_ns_per_update=[math]::Round($nns,3);ns_change_percent=[math]::Round(100*($nns/$pns-1),3);parent_cycles_per_update=[math]::Round($pc,3);current_cycles_per_update=[math]::Round($nc,3);cycles_change_percent=[math]::Round(100*($nc/$pc-1),3);parent_ns_drift_percent=[math]::Round(100*($pn2/$pn1-1),3);parent_cycles_drift_percent=[math]::Round(100*($pc2/$pc1-1),3)}
}
$summary | Export-Csv -LiteralPath (Join-Path $results 'shadow-mask-cost-word-summary14m.csv') -NoTypeInformation
$summary | ConvertTo-Csv -NoTypeInformation
