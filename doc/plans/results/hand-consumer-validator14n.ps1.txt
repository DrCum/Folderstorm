$ErrorActionPreference='Stop'

 @{source='hand-consumer-current14n.cpp';name='hand-consumer-current-map14n';defs=@();args=@('--trace')},
 @{source='hand-consumer-current14n.cpp';name='hand-consumer-current-contiguous14n';defs=@('FS_CONTIGUOUS_ANIMATION_KEYS=1');args=@('--trace')},

 }
}
